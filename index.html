/* * Whiteout Survival — Isometric Tile Ground * Terreno-base para o projeto Three.js. * * Referência de grade: * https://github.com/trevorscottprice/wos-tile-mosaic * * IMPORTANTE: * Este terreno é provisório. Não contém as texturas * originais extraídas do Whiteout Survival. */ export function createTileGround(THREE, options = {}) { const config = { width: 62, height: 62, tileSize: 1, seed: 2026, ...options }; const width = Math.max(1, Math.floor(config.width)); const height = Math.max(1, Math.floor(config.height)); const tileSize = Math.max(0.01, config.tileSize); const count = width * height; const group = new THREE.Group(); group.name = "WhiteoutCityGround"; // Gerador determinístico: o mesmo seed produz a mesma grade. let seed = config.seed >>> 0; function random() { seed = (1664525 * seed + 1013904223) >>> 0; return seed / 4294967296; } // Uma geometria quadrada compartilhada por todas as peças. const geometry = new THREE.PlaneGeometry( tileSize, tileSize ); geometry.rotateX(-Math.PI / 2); const materials = [ new THREE.MeshStandardMaterial({ color: 0xe8edf0, roughness: 1 }), new THREE.MeshStandardMaterial({ color: 0xd7e0e7, roughness: 1 }), new THREE.MeshStandardMaterial({ color: 0xc6d2dc, roughness: 1 }), new THREE.MeshStandardMaterial({ color: 0xf5f6f7, roughness: 1 }) ]; // Cada material possui sua própria lista de instâncias. const meshes = materials.map((material, i) => { const mesh = new THREE.InstancedMesh( geometry, material, count ); mesh.name = `SnowGroundTiles_${i}`; mesh.receiveShadow = true; mesh.castShadow = false; group.add(mesh); return mesh; }); const dummy = new THREE.Object3D(); const used = [0, 0, 0, 0]; // A grade fica centralizada na origem. for (let z = 0; z < height; z++) { for (let x = 0; x < width; x++) { dummy.position.set( (x - (width - 1) / 2) * tileSize, 0, (z - (height - 1) / 2) * tileSize ); dummy.rotation.set(0, 0, 0); dummy.scale.set(1, 1, 1); dummy.updateMatrix(); const value = random(); const materialIndex = value < 0.55 ? 0 : value < 0.78 ? 1 : value < 0.92 ? 2 : 3; const mesh = meshes[materialIndex]; // Índice independente para cada material. mesh.setMatrixAt( used[materialIndex], dummy.matrix ); used[materialIndex]++; } } // Define a quantidade real de instâncias usadas em cada mesh. for (let i = 0; i < meshes.length; i++) { const mesh = meshes[i]; mesh.count = used[i]; mesh.instanceMatrix.needsUpdate = true; mesh.computeBoundingBox(); mesh.computeBoundingSphere(); } group.userData.grid = { width, height, tileSize, seed: config.seed, totalTiles: count }; // Facilita a remoção posterior do terreno da cena. group.userData.dispose = function () { geometry.dispose(); for (const material of materials) { material.dispose(); } }; return group; }
