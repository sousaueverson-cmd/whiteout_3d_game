/* * Whiteout Survival — Isometric Tile Ground * Base de terreno para o projeto Three.js. * * Inspirado no método de grade isométrica do: * https://github.com/trevorscottprice/wos-tile-mosaic * * Este arquivo gera terreno provisório. * Para reproduzir o terreno original, substitua as cores * pelas texturas extraídas dos arquivos do jogo. */ export function createTileGround(THREE, options = {}) { const config = { width: options.width ?? 62, height: options.height ?? 62, tileSize: options.tileSize ?? 1, seed: options.seed ?? 2026, ...options }; const group = new THREE.Group(); group.name = "WhiteoutCityGround"; const random = (() => { let seed = config.seed >>> 0; return () => { seed = (1664525 * seed + 1013904223) >>> 0; return seed / 4294967296; }; })(); const geometry = new THREE.PlaneGeometry( config.tileSize, config.tileSize ); geometry.rotateX(-Math.PI / 2); const materials = [ new THREE.MeshStandardMaterial({ color: 0xe8edf0, roughness: 1 }), new THREE.MeshStandardMaterial({ color: 0xd7e0e7, roughness: 1 }), new THREE.MeshStandardMaterial({ color: 0xc6d2dc, roughness: 1 }), new THREE.MeshStandardMaterial({ color: 0xf5f6f7, roughness: 1 }) ]; const count = config.width * config.height; const meshes = materials.map(material => { const mesh = new THREE.InstancedMesh( geometry, material, count ); mesh.name = "SnowGroundTiles"; mesh.receiveShadow = true; mesh.castShadow = false; group.add(mesh); return mesh; }); const dummy = new THREE.Object3D(); const tile = config.tileSize; let index = 0; for (let z = 0; z < config.height; z++) { for (let x = 0; x < config.width; x++) { // Grade regular; a câmera pode aplicar a perspectiva // isométrica ao conjunto inteiro. dummy.position.set( (x - (config.width - 1) / 2) * tile, 0, (z - (config.height - 1) / 2) * tile ); dummy.updateMatrix(); const variation = random(); const materialIndex = variation < 0.55 ? 0 : variation < 0.78 ? 1 : variation < 0.92 ? 2 : 3; const mesh = meshes[materialIndex]; mesh.setMatrixAt(index, dummy.matrix); index++; } } for (const mesh of meshes) { mesh.instanceMatrix.needsUpdate = true; } group.userData.grid = { width: config.width, height: config.height, tileSize: tile, seed: config.seed }; return group; }
